```python
import argparse
import json
from search_r1.llm_agent import LLMGenerationManager

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--model_path", type=str, required=True)
    parser.add_argument("--input_file", type=str, required=True)
    parser.add_argument("--retriever_url", type=str, default="http://127.0.0.1:8000/retrieve")
    parser.add_argument("--max_turns", type=int, default=8)
    parser.add_argument("--output_file", type=str, default="results.json")
    args = parser.parse_args()
    
    # 初始化 Agent
    manager = LLMGenerationManager(
        model_path=args.model_path,
        retriever_url=args.retriever_url,
        max_turns=args.max_turns,
    )
    
    # 加载问题
    with open(args.input_file, 'r') as f:
        questions = [json.loads(line)["question"] for line in f]
    
    # 批量���理
    results = manager.run_llm_loop(questions)
    
    # 保存结果
    with open(args.output_file, 'w') as f:
        for r in results:
            f.write(json.dumps(r) + '\n')
    
    print(f"Saved {len(results)} results to {args.output_file}")

if __name__ == "__main__":
    main()
```

## 9.3 评估指标

### 9.3.1 精确匹配 (Exact Match, EM)

```python
def em_score(predictions: List[str], ground_truths: List[List[str]]) -> float:
    """
    计算精确匹配分数
    
    Args:
        predictions: 预测答案列表
        ground_truths: 标准答案列表（每个问题可能有多个标准答案）
        
    Returns:
        EM 分数 (0-1)
    """
    correct = 0
    for pred, gts in zip(predictions, ground_truths):
        pred_normalized = normalize_answer(pred)
        for gt in gts:
            if pred_normalized == normalize_answer(gt):
                correct += 1
                break
    return correct / len(predictions)
```

### 9.3.2 F1 分数

```python
def f1_score(prediction: str, ground_truths: List[str]) -> float:
    """
    计算 F1 分数（词级别）
    """
    def get_tokens(s):
        return set(s.lower().split())
    
    pred_tokens = get_tokens(prediction)
    best_f1 = 0
    
    for gt in ground_truths:
        gt_tokens = get_tokens(gt)
        
        common = pred_tokens & gt_tokens
        if len(common) == 0:
            continue
            
        precision = len(common) / len(pred_tokens)
        recall = len(common) / len(gt_tokens)
        f1 = 2 * precision * recall / (precision + recall)
        best_f1 = max(best_f1, f1)
    
    return best_f1
```

### 9.3.3 完整评估脚本

```python
def evaluate(predictions_file: str, ground_truth_file: str) -> dict:
    """
    完整评估
    """
    # 加载预测结果
    predictions = []
    with open(predictions_file, 'r') as f:
        for line in f:
            data = json.loads(line)
            predictions.append(extract_answer(data["response"]))
    
    # 加载标准答案
    ground_truths = []
    with open(ground_truth_file, 'r') as f:
        for line in f:
            data = json.loads(line)
            ground_truths.append(data["answers"])
    
    # 计算指标
    em = em_score(predictions, ground_truths)
    f1 = f1_score_avg(predictions, ground_truths)
    
    return {
        "exact_match": em,
        "f1": f1,
        "num_samples": len(predictions),
    }
```

```python
def batch_inference(manager: LLMGenerationManager, questions: List[str], batch_size: int = 16):
    """
    批量推理
    """
    results = []
    for i in range(0, len(questions), batch_size):
        batch = questions[i:i+batch_size]
        batch_results = manager.run_llm_loop(batch)
        results.extend(batch_results)
        print(f"Processed {min(i+batch_size, len(questions))}/{len(questions)}")
    return results
```

## 9.7 错误分析

### 9.7.1 常见错误类型

| 错误类型 | 描述 | 解决方案 |
|----------|------|----------|
| 未搜索 | 模型不调用搜索 | 增加搜索奖励 |
| 错误搜索 | 搜索不相关查询 | 改进 prompt |
| 错误答案 | 答案与事实不符 | 增加训练数据 |
| 格式错误 | 未输出 `<answer>` | 改进格式提示 |

### 9.7.2 错误分析脚本

```python
def analyze_errors(results: List[dict], ground_truths: List[dict]):
    """
    错误分析
    """
    errors = {
        "no_search": [],      # 没有搜索的情况
        "wrong_query": [],   # 搜索了但查询错误
        "wrong_answer": [],   # 搜索了但答案错误
        "format_error": [],   # 格式错误
    }
    
    for i, (result, gt) in enumerate(zip(results, ground_truths)):
        response = result["response"]
        
        # 检查是否有搜索
        if "<search>" not in response:
            errors["no_search"].append(i)
        
        # 检查答案
        answer = extract_answer(response)
        if answer and not em_check(answer, gt["target"]):
            errors["wrong_answer"].append(i)
    
    # 统计
    print("=== Error Analysis ===")
    for error_type, indices in errors.items():
        print(f"{error_type}: {len(indices)} ({len(indices)/len(results)*100:.1f}%)")
    
    return errors
```

```python
def plot_accuracy_vs_turns(results: List[dict]):
    """绘制准确率随轮次变化"""
    turns_accuracies = {}
    
    for r in results:
        turns = r["num_turns"]
        correct = r.get("correct", False)
        if turns not in turns_accuracies:
            turns_accuracies[turns] = []
        turns_accuracies[turns].append(correct)
    
    x = sorted(turns_accuracies.keys())
    y = [mean(turns_accuracies[t]) for t in x]
    
    plt.plot(x, y, marker='o')
    plt.xlabel("Number of Turns")
    plt.ylabel("Accuracy")
    plt.title("Accuracy vs Number of Turns")
    plt.savefig("accuracy_vs_turns.png")
```
