## 7.1 数据概述

Search-R1 使用两类主要数据：
1. **训练数据**：包含问题、答案、提示模板
2. **语料库**：用于检索的文档集合

## 7.2 数据格式详解

### 7.2.1 训练数据格式 (parquet)

```python
{
    "data_source": str,      # 数据源标识，如 "nq", "hotpotqa"
    "prompt": List[dict],   # 对话格式的提示
    "ability": str,         # 能力类型，如 "fact-reasoning"
    "reward_model": {
        "style": "rule",    # 奖励风格，"rule" 表示基于规则
        "ground_truth": {
            "target": List[str]  # 标准答案列表
        }
    },
    "extra_info": {
        "split": str,       # "train" 或 "val"
        "index": int        # 数据索引
    }
}
```

### 7.2.2 提示模板格式

```python
prompt = [
    {

        "content": "You are a helpful assistant. You can use <search>...</search> to search for information, and <answer>...</answer> to provide the final answer."

        "role": "user",
        "content": "Question: {question}\nPlease answer and show your reasoning process."
    }
]
```

### 7.2.3 语料库格式 (JSONL)

```json
{"id": "0", "contents": "标题\\n正文内容..."}
{"id": "1", "contents": "另一个标题\\n更多内容..."}
```

## 7.3 数据处理脚本

### 7.3.1 NQ Search 数据处理

文件：`scripts/data_process/nq_search.py`

```python
import pandas as pd
from pathlib import Path

def process_nq_search(input_file: str, output_dir: str, max_samples: int = None):
    """
    处理 Natural Questions 数据集
    
    Args:
        input_file: 原始 JSON 文件路径
        output_dir: 输出目录
        max_samples: 最大样本数
    """
    # 读取原始数据
    with open(input_file, 'r') as f:
        data = [json.loads(line) for line in f]
    
    if max_samples:
        data = data[:max_samples]
    
    # 转换为训练格式
    processed = []
    for item in data:
        example = {
            "data_source": "nq",
            "prompt": [
                {
                    "role": "system",
                    "content": "You are a helpful assistant..."
                },
                {
                    "role": "user",
                    "content": f"Question: {item['question']}"
                }
            ],
            "ability": "fact-reasoning",
            "reward_model": {
                "style": "rule",
                "ground_truth": {
                    "target": item.get("answers", [])
                }
            },
            "extra_info": {
                "split": item.get("split", "train"),
                "index": len(processed)
            }
        }
        processed.append(example)
    
    # 保存为 parquet
    df = pd.DataFrame(processed)
    output_path = Path(output_dir)
    output_path.mkdir(parents=True, exist_ok=True)
    
    df.to_parquet(output_path / "train.parquet")
    print(f"Saved {len(df)} samples to {output_path / 'train.parquet'}")
```

## 7.5 数据验证

### 7.5.1 验证数据格式

```python
import pandas as pd

def validate_data(parquet_path: str) -> bool:
    """验证 parquet 数据文件格式"""
    df = pd.read_parquet(parquet_path)
    
    required_columns = ["data_source", "prompt", "ability", "reward_model", "extra_info"]
    for col in required_columns:
        if col not in df.columns:
            print(f"Missing column: {col}")
            return False
    
    # 验证每个样本
    for idx, row in df.iterrows():
        if not isinstance(row["prompt"], list):
            print(f"Row {idx}: prompt is not a list")
            return False
        if "ground_truth" not in row["reward_model"]:
            print(f"Row {idx}: missing ground_truth")
            return False
    
    print(f"Validation passed! {len(df)} samples.")
    return True
```

```python
import pandas as pd

def analyze_data(parquet_path: str):
    df = pd.read_parquet(parquet_path)
    
    print("=== 数据统计 ===")
    print(f"总样本数: {len(df)}")
    print(f"数据源分布: {df['data_source'].value_counts().to_dict()}")
    print(f"能力类型分布: {df['ability'].value_counts().to_dict()}")
    
    # 检查答案长度
    answer_lens = []
    for _, row in df.iterrows():
        targets = row["reward_model"]["ground_truth"]["target"]
        answer_lens.extend([len(a) for a in targets])
    
    print(f"答案长度: mean={sum(answer_lens)/len(answer_lens):.1f}, "
          f"max={max(answer_lens)}")
```

## 7.6 自定义数据准备

### 7.6.1 创建自定义数据集

```python
import json

def create_custom_dataset(
    questions: List[str],
    answers: List[List[str]],
    output_path: str
):
    """
    创建自定义训练数据集
    
    Args:
        questions: 问题列表
        answers: 对应答案列表（每个问题可能有多个标准答案）
        output_path: 输出文件路径
    """
    data = []
    for q, ans in zip(questions, answers):
        example = {
            "data_source": "custom",
            "prompt": [
                {
                    "role": "system",
                    "content": "You are a helpful assistant..."
                },
                {
                    "role": "user",
                    "content": f"Question: {q}"
                }
            ],
            "ability": "fact-reasoning",
            "reward_model": {
                "style": "rule",
                "ground_truth": {
                    "target": ans
                }
            },
            "extra_info": {
                "split": "train",
                "index": len(data)
            }
        }
        data.append(example)
    
    # 保存为 parquet
    import pandas as pd
    df = pd.DataFrame(data)
    df.to_parquet(output_path)
    
    print(f"Created dataset with {len(data)} samples")
```

### 7.6.2 格式转换脚本

```python
# CSV 转 parquet
import pandas as pd

def csv_to_parquet(csv_path: str, output_path: str):
    """将 CSV 转换为 parquet 格式"""
    df = pd.read_csv(csv_path)
    
    # 假设 CSV 有 question, answer 列
    df["prompt"] = df["question"].apply(lambda x: [
        {"role": "user", "content": f"Question: {x}"}
    ])
    df["reward_model"] = df["answer"].apply(lambda x: {
        "style": "rule",
        "ground_truth": {"target": [x]}
    })
    
    df.to_parquet(output_path)
```

## 7.7 数据增强策略

### 7.7.1 提示变体

```python
PROMPT_VARIANTS = [
    "Question: {q}\nPlease answer.",
    "Question: {q}\nThink carefully and provide your answer.",
    "{q}\nWhat is the answer?",
    "Help me answer this: {q}",
]

def augment_prompts(data: List[dict]) -> List[dict]:
    """通过改变提示模板来增强数据"""
    augmented = []
    for item in data:
        question = item["prompt"][-1]["content"].replace("Question: ", "")
        for template in PROMPT_VARIANTS:
            new_item = item.copy()
            new_item["prompt"][-1]["content"] = template.format(q=question)
            augmented.append(new_item)
    return augmented
```
