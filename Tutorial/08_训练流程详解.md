## 8.3 训练循环详解

### 8.3.1 RayPPOTrainer.fit() 主循环

```python
def fit(self):
    """主训练循环"""
    
    for epoch in range(self.config.trainer.total_epochs):
        for step, batch in enumerate(self.dataloader):
            
            # ===== Phase 1: 生成阶段 =====
            if self.config.agent.do_search:
                # 使用 Agent 生成（带搜索）
                samples = self._generate_with_search(batch)
            else:
                # 标准生成（无搜索）
                samples = self._generate_standard(batch)
            
            # ===== Phase 2: 奖励计算 =====
            rewards = self._compute_rewards(samples)
            samples["rewards"] = rewards
            
            # ===== Phase 3: KL 惩罚 =====
            if self.config.algorithm.use_kl_penalty:
                samples = self._apply_kl_penalty(samples)
            
            # ===== Phase 4: 优势估计 =====
            if self.config.algorithm.adv_estimator == "gae":
                # GAE 方法（需要 Critic）
                values = self.critic(samples["states"])
                advantages, returns = compute_gae_advantage(
                    values, rewards,
                    gamma=self.config.algorithm.gamma,
                    lam=self.config.algorithm.lam
                )
            else:
                # GRPO 方法（组内相对）
                advantages = compute_grpo_outcome_advantage(rewards)
                returns = rewards  # GRPO 使用原始奖励作为回报
            
            samples["advantages"] = advantages
            samples["returns"] = returns
            
            # ===== Phase 5: 更新 Critic（仅 GAE）=====
            if self.config.algorithm.adv_estimator == "gae":
                for _ in range(self.config.trainer.critic_update_epochs):
                    critic_loss = self._update_critic(samples)
            
            # ===== Phase 6: 更新 Actor =====
            for _ in range(self.config.trainer.actor_update_epochs):
                actor_loss = self._update_actor(samples)
            
            # ===== Phase 7: 验证和保存 =====
            if step % self.config.trainer.val_freq == 0:
                self._validate()
            
            if step % self.config.trainer.save_freq == 0:
                self._save_checkpoint(step)
```

## 8.4 生成阶段详解

### 8.4.1 带搜索的生成 (_generate_with_search)

```python
def _generate_with_search(self, batch: dict) -> dict:
    """
    使用 LLMGenerationManager 生成带搜索的样本
    """
    # 1. 准备 prompts
    prompts = []
    for item in batch:
        prompt = self._format_prompt(item)
        prompts.append(prompt)
    
    # 2. 调用 Agent 生成（多轮循环）
    samples = self.agent_manager.run_llm_loop(prompts)
    
    # 3. 提取关键信息
    return {
        "input_ids": samples["input_ids"],
        "response_ids": samples["response_ids"],
        "logprobs": samples["logprobs"],
        "responses": samples["responses"],
        "search_queries": samples["search_queries"],
        "num_turns": samples["num_turns"],
    }
```

### 8.4.2 LLMGenerationManager.run_llm_loop()

```python
def run_llm_loop(self, prompts: List[str]) -> List[dict]:
    """
    多轮推理循环的核心实现
    """
    states = []  # 存储每个样本的状态
    
    for prompt in prompts:
        states.append({
            "prompt": prompt,
            "history": [],        # 对话历史
            "search_count": 0,    # 搜索次数
            "done": False,
        })
    
    # 迭代推理
    for turn in range(self.max_turns):
        # 找出未完成的样本
        active_indices = [i for i, s in enumerate(states) if not s["done"]]
        if not active_indices:
            break
        
        # 准备输入
        inputs = []
        for idx in active_indices:
            full_prompt = self._build_prompt(
                states[idx]["prompt"],
                states[idx]["history"]
            )
            inputs.append(full_prompt)
        
        # 调用 LLM 生成
        outputs = self.rollout.generate(inputs)
        
        # 处理每个输出
        for i, idx in enumerate(active_indices):
            output = outputs[i]
            states[idx]["history"].append(output)
            
            # 解析动作
            action = self._parse_action(output)
            
            if action["type"] == "search":
                # 执行搜索
                query = action["query"]
                results = self._do_search(query)
                states[idx]["history"].append(f"Search results: {results}")
                states[idx]["search_count"] += 1
                
            elif action["type"] == "answer":
                # 任务完成
                states[idx]["done"] = True
                states[idx]["answer"] = action["answer"]
    
    return states
```

### 8.5.1 奖励计算流程

```python
def _compute_rewards(self, samples: dict) -> torch.Tensor:
    """
    计算奖励分数
    """
    rewards = []
    
    for i in range(len(samples["responses"])):
        response = samples["responses"][i]
        
        # 提取 <answer> 标签中的答案
        answer = extract_answer(response)
        
        # 获取 ground truth
        ground_truth = samples["ground_truth"][i]
        
        # 计算 EM 分数
        score = compute_score_em(response, ground_truth)
        rewards.append(score)
    
    return torch.tensor(rewards, device=self.device)
```

### 8.5.2 奖励函数 (qa_em.py)

```python
def compute_score_em(
    solution_str: str,
    ground_truth: dict,
    format_score: float = 0.,
    score: float = 1.,
) -> float:
    """
    精确匹配评分函数
    
    从 solution_str 中提取 <answer> 标签，
    与 ground_truth['target'] 比较。
    """
    # 提取答案
    answer = extract_answer(solution_str)
    
    if answer is None:
        return format_score  # 没有有效答案
    
    # 检查是否匹配
    targets = ground_truth.get('target', [])
    for target in targets:
        if normalize_answer(answer) == normalize_answer(target):
            return score
    
    return format_score  # 格式正确但答案错误
```

## 8.6 优势估计详解

### 8.6.1 GAE 优势估计

```python
def compute_gae_advantage_return(
    values: torch.Tensor,
    rewards: torch.Tensor,
    gamma: float = 0.99,
    lam: float = 0.95,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    计算 GAE 优势估计
    
    GAE 平衡了 TD(λ) 的偏差和方差：
    - λ=0: 单步 TD（高偏差，低方差）
    - λ=1: 多步蒙特卡洛（低偏差，高方差）
    """
    batch_size, seq_len = rewards.shape
    advantages = torch.zeros_like(values)
    
    gae = 0
    for t in reversed(range(seq_len)):
        if t == seq_len - 1:
            next_value = 0  # 终止状态
        else:
            next_value = values[:, t + 1]
        
        # TD 误差
        delta = rewards[:, t] + gamma * next_value - values[:, t]
        
        # 累积
        gae = delta + gamma * lam * gae
        advantages[:, t] = gae
    
    # 回报 = 优势 + 价值
    returns = advantages + values
    
    return advantages, returns
```

### 8.6.2 GRPO 优势计算

```python
def compute_grpo_outcome_advantage(
    rewards: torch.Tensor,
) -> torch.Tensor:
    """
    GRPO 优势计算（组内相对）
    
    GRPO 不需要 Critic 模型，使用组内相对基准：
    - 同一个 prompt 生成的多个样本组成一个组
    - 优势 = (r - μ) / σ
    """
    # rewards shape: [num_prompts, samples_per_prompt]
    batch_size = rewards.shape[0]
    
    # 计算组内均值和标准差
    group_mean = rewards.mean(dim=-1, keepdim=True)
    group_std = rewards.std(dim=-1, keepdim=True) + 1e-8
    
    # 组内相对优势
    advantages = (rewards - group_mean) / group_std
    
    return advantages
```

## 8.7 策略更新详解

### 8.7.1 PPO 策略损失

```python
def compute_policy_loss(
    logprobs: torch.Tensor,
    old_logprobs: torch.Tensor,
    advantages: torch.Tensor,
    epsilon: float = 0.2,
) -> torch.Tensor:
    """
    计算 PPO CLIP 损失
    """
    # 重要性采样比率
    ratio = torch.exp(logprobs - old_logprobs)
    
    # Clipped 目标
    surr1 = ratio * advantages
    surr2 = torch.clamp(ratio, 1 - epsilon, 1 + epsilon) * advantages
    
    # 取较小值（悲观估计）
    loss = -torch.min(surr1, surr2).mean()
    
    return loss
```

### 8.7.2 价值损失

```python
def compute_value_loss(
    values: torch.Tensor,
    returns: torch.Tensor,
    old_values: torch.Tensor,
    clip_epsilon: float = 0.2,
) -> torch.Tensor:
    """
    计算价值函数损失
    """
    # Clipped value error
    values_clipped = old_values + torch.clamp(
        values - old_values,
        -clip_epsilon,
        clip_epsilon
    )
    
    # MSE loss
    loss1 = (values - returns) ** 2
    loss2 = (values_clipped - returns) ** 2
    
    # 取较大值
    loss = torch.max(loss1, loss2).mean()
    
    return loss
```

def _save_checkpoint(self, step: int):
    """保存模型检查点"""
    ckpt_path = f"./checkpoints/step_{step}"
    
    # 保存 Actor 模型
    self.actor.save(ckpt_path / "actor")

## 8.11 本章小结

### 训练流程总结
1. **数据加载**：从 parquet 文件加载 batch
2. **生成样本**：调用 LLM 生成响应（可能多轮搜索）
3. **奖励计算**：基于答案正确性计算奖励
4. **优势估计**：GAE 或 GRPO
5. **策略更新**：PPO CLIP 损失
6. **验证保存**：定期验证和保存

### 常见问题解决
- 奖励为 0 → 检查答案提取逻辑
- 训练不稳定 → 降低学习率，检查 KL 系数
- 搜索过多 → 调整 max_turns 或奖励结构
