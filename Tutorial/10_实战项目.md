```python
class CustomRetriever:
    def __init__(self, config):
        self.config = config
        
    def retrieve(self, query: str, topk: int = 3) -> List[dict]:
        # 1. 编码查询
        query_embedding = self.encoder.encode(query)
        
        # 2. 搜索索引
        results = self.index.search(query_embedding, topk)
        
        # 3. 后处理
        return self.postprocess(results)
    
    def postprocess(self, results: List[dict]) -> List[dict]:
        # 添加摘要、重排序等
        return results
```

### 10.4.3 复现关键代码

```python
# GRPO 核心实现
def compute_grpo_advantage(rewards):
    """
    GRPO 优势计算
    来自 DeepSeek-R1
    """
    group_size = 8  # 每个 prompt 生成 8 个样本
    num_groups = len(rewards) // group_size
    
    advantages = []
    for i in range(num_groups):
        group_rewards = rewards[i*group_size:(i+1)*group_size]
        mean = sum(group_rewards) / len(group_rewards)
        std = (sum((r - mean)**2 for r in group_rewards) / len(group_rewards))**0.5
        std = max(std, 1e-8)
        
        for r in group_rewards:
            advantages.append((r - mean) / std)
    
    return torch.tensor(advantages)
```

### 10.5.1 测试脚本

```python
def benchmark(model_path: str, dataset: str):
    """
    基准测试
    """
    # 加载模型
    manager = LLMGenerationManager(model_path=model_path)
    
    # 测试数据
    test_questions = load_dataset(dataset)
    
    # 计时推理
    import time
    start = time.time()
    results = manager.run_llm_loop(test_questions)
    elapsed = time.time() - start
    
    # 计算指标
    metrics = evaluate(results)
    metrics["time"] = elapsed
    metrics["throughput"] = len(test_questions) / elapsed
    
    return metrics
```

```python
from search_r1.llm_agent import LLMGenerationManager

class SearchR1Agent(Agent):
    def __init__(self, model_path, retriever_url):
        self.manager = LLMGenerationManager(model_path, retriever_url)
    
    def run(self, input_str):
        return self.manager.run_llm_loop([input_str])
    
    @property
    def tools(self):
        return [self.retriever]
```

### 10.6.2 与 RAG 系统集成

```python
class RAGWithSearchAgent:
    """
    结合传统 RAG 和 Search-R1 Agent
    """
    def __init__(self, retriever, generator):
        self.retriever = retriever
        self.generator = generator
    
    def answer(self, question):
        # 1. 检索相关文档
        docs = self.retriever.search(question, topk=5)
        
        # 2. 构建增强 prompt
        context = "\n".join([d["content"] for d in docs])
        prompt = f"Context: {context}\nQuestion: {question}"
        
        # 3. 使用 Agent 生成（可能进行额外搜索）
        result = self.generator.run_llm_loop([prompt])
        
        return result[0]
```
