## 4.1 generation.py 详解

文件路径：`search_r1/llm_agent/generation.py`

### 核心类：LLMGenerationManager

```python
class LLMGenerationManager:
    """
    LLM Agent 生成管理器
    负责管理 LLM 的多轮推理循环，包括：
    1. 调用 LLM 生成
    2. 解析生成结果中的搜索动作
    3. 执行搜索 API
    4. 将搜索结果加入上下文
    """
    
    def __init__(
        self,
        rollout_worker,      # vLLM rollout worker
        tokenizer,           # 分词器
        retriever_url,       # 检索服务器 URL
        max_turns: int = 16,  # 最大推理轮次
        do_search: bool = True,
    ):
        self.rollout = rollout_worker
        self.tokenizer = tokenizer
        self.retriever_url = retriever_url
        self.max_turns = max_turns
        self.do_search = do_search
```

### 关键方法解析

#### 4.1.1 run_llm_loop() - 主循环

```python
def run_llm_loop(self, prompts: List[str]) -> List[dict]:
    """
    执行多轮推理循环
    
    Args:
        prompts: 输入提示列表
        
    Returns:
        包含生成结果的字典列表
    """
    # 初始化状态
    states = [AgentState(p) for p in prompts]
    
    # 迭代推理
    for turn in range(self.max_turns):
        # 1. 为未完成的样本生成
        active_indices = [i for i, s in enumerate(states) if not s.done]
        if not active_indices:
            break
            
        # 2. 构造输入
        inputs = [states[i].get_input() for i in active_indices]
        
        # 3. 调用 LLM 生成
        outputs = self.rollout.generate(inputs)
        
        # 4. 处理输出
        for idx, output in zip(active_indices, outputs):
            states[idx].add_generation(output)
            
        # 5. 解析和执行动作
        if self.do_search:
            self._process_searches(states, active_indices)
        
        # 6. 检查是否全部完成
        if all(s.done for s in states):
            break
            
    return [s.to_dict() for s in states]
```

#### 4.1.2 execute_predictions() - 执行预测

```python
def execute_predictions(
    self,
    input_ids: torch.Tensor,
    prompts: List[str],
) -> Dict[str, Any]:
    """
    执行批量预测，处理搜索动作
    
    这是被 verl 框架调用的主入口
    """
    # 1. 调用 run_llm_loop
    results = self.run_llm_loop(prompts)
    
    # 2. 整理输出格式
    output = {
        "sequences": [],      # 完整序列
        "responses": [],     # 响应部分
        "logprobs": [],      # 对数概率
        "search_queries": [], # 搜索查询
    }
    
    for r in results:
        output["sequences"].append(r["sequence"])
        output["responses"].append(r["response"])
        output["search_queries"].append(r["queries"])
        
    return output
```

#### 4.1.3 batch_search() - 批量搜索

```python
def batch_search(self, queries: List[str]) -> List[str]:
    """
    批量调用检索服务
    
    Args:
        queries: 搜索查询列表
        
    Returns:
        检索结果列表
    """
    import aiohttp
    import asyncio
    
    async def _search_single(query):
        async with aiohttp.ClientSession() as session:
            async with session.post(
                self.retriever_url,
                json={"query": query, "topk": 3},
                timeout=aiohttp.ClientTimeout(total=30)
            ) as resp:
                data = await resp.json()
                return data["results"]
                
    # 并发执行所有搜索
    loop = asyncio.new_event_loop()
    results = loop.run_until_complete(
        asyncio.gather(*[_search_single(q) for q in queries])
    )
    loop.close()
    
    return results
```

#### 4.1.4 postprocess_predictions() - 后处理

```python
def postprocess_predictions(self, outputs: List[str]) -> Dict:
    """
    解析输出中的 <search> 和 <answer> 标签
    """
    import re
    
    results = []
    for output in outputs:
        # 提取搜索动作
        search_pattern = r'<search>(.*?)</search>'
        searches = re.findall(search_pattern, output)
        
        # 提取答案
        answer_pattern = r'<answer>(.*?)</answer>'
        answers = re.findall(answer_pattern, output)
        
        # 移除所有标签后的纯文本
        clean_text = re.sub(r'<.*?>', '', output)
        
        results.append({
            "raw": output,
            "clean": clean_text,
            "searches": searches,
            "answer": answers[-1] if answers else None,
        })
        
    return results
```

### 关键方法解析

#### 4.2.1 fit() - 主训练循环

```python
def fit(self):
    """
    主训练循环
    """
    for epoch in range(self.config.trainer.total_epochs):
        for batch in self.dataloader:
            # ===== 生成阶段 =====
            # 如果启用搜索，使用 Agent 生成
            if self.config.agent.do_search:
                samples = self._generate_with_search(batch)
            else:
                samples = self._generate_standard(batch)
                
            # ===== 奖励计算 =====
            rewards = self._compute_rewards(samples)
            samples["rewards"] = rewards
            
            # ===== 应用 KL 惩罚 =====
            if self.config.algorithm.use_kl_penalty:
                samples = self._apply_kl_penalty(samples)
                
            # ===== 计算优势函数 =====
            if self.config.algorithm.adv_estimator == "gae":
                advantages = compute_gae_advantage(samples)
            else:  # grpo
                advantages = compute_grpo_outcome_advantage(samples)
            samples["advantages"] = advantages
            
            # ===== 更新 Critic =====
            for _ in range(self.config.trainer.critic_update_epochs):
                self._update_critic(samples)
                
            # ===== 更新 Actor =====
            for _ in range(self.config.trainer.actor_update_epochs):
                self._update_actor(samples)
                
            # ===== 验证和保存 =====
            if self.step % self.config.trainer.val_freq == 0:
                self._validate()
            if self.step % self.config.trainer.save_freq == 0:
                self._save_checkpoint()
```

## 4.3 core_algos.py 详解

### 核心函数

#### 4.3.1 compute_gae_advantage() - GAE 优势估计

```python
def compute_gae_advantage_return(
    values: torch.Tensor,           # V(s_t)
    rewards: torch.Tensor,          # r_t
    gamma: float = 0.99,            # 折扣因子
    lam: float = 0.95,              # GAE lambda
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    计算 GAE 优势估计和回报
    
    Args:
        values: 价值估计 [batch_size, seq_len]
        rewards: 奖励 [batch_size, seq_len]
        gamma: 折扣因子
        lam: GAE lambda 参数
        
    Returns:
        advantages: 优势 [batch_size, seq_len]
        returns: 回报 [batch_size, seq_len]
    """
    advantages = torch.zeros_like(values)
    
    # 反向计算 GAE
    gae = 0
    for t in reversed(range(len(rewards))):
        if t == len(rewards) - 1:
            next_value = 0  # 终止状态
        else:
            next_value = values[:, t + 1]
            
        delta = rewards[:, t] + gamma * next_value - values[:, t]
        gae = delta + gamma * lam * gae
        advantages[:, t] = gae
        
    # 计算回报 = 优势 + 价值
    returns = advantages + values
    
    return advantages, returns
```

#### 4.3.2 compute_grpo_outcome_advantage() - GRPO 优势计算

```python
def compute_grpo_outcome_advantage(
    rewards: torch.Tensor,          # [batch_size, group_size]
) -> torch.Tensor:
    """
    计算 GRPO 优势（组内相对基准）
    
    GRPO 将同一 prompt 的多个样本组成一个组，
    用组内相对表现计算优势。
    """
    # rewards shape: [num_prompts, samples_per_prompt]
    # 每个 prompt 有 samples_per_prompt 个样本
    
    group_mean = rewards.mean(dim=-1, keepdim=True)  # 组均值
    group_std = rewards.std(dim=-1, keepdim=True)    # 组标准差
    
    # 优势 = (r - μ) / σ
    advantages = (rewards - group_mean) / (group_std + 1e-8)
    
    return advantages
```

#### 4.3.3 compute_policy_loss() - PPO 策略损失

```python
def compute_policy_loss(
    logprobs: torch.Tensor,          # 新策略 log π(a|s)
    old_logprobs: torch.Tensor,      # 旧策略 log π_old(a|s)
    advantages: torch.Tensor,        # 优势估计 A(s,a)
    epsilon: float = 0.2,            # clip 范围
) -> torch.Tensor:
    """
    计算 PPO CLIP 损失
    """
    # 计算重要性采样比率
    ratio = torch.exp(logprobs - old_logprobs)
    
    # 计算 clipped 目标
    surr1 = ratio * advantages
    surr2 = torch.clamp(ratio, 1 - epsilon, 1 + epsilon) * advantages
    
    # 取较小值（悲观估计）
    policy_loss = -torch.min(surr1, surr2).mean()
    
    return policy_loss
```

#### 4.3.4 compute_value_loss() - 价值损失

```python
def compute_value_loss(
    values: torch.Tensor,           # 预测价值
    returns: torch.Tensor,           # 目标回报
    old_values: torch.Tensor,       # 旧价值估计
    clip_value_epsilon: float = 0.2,
) -> torch.Tensor:
    """
    计算价值函数损失（带 clip）
    """
    # Clipped value error
    values_clipped = old_values + torch.clamp(
        values - old_values,
        -clip_value_epsilon,
        clip_value_epsilon
    )
    
    # MSE loss
    loss1 = (values - returns) ** 2
    loss2 = (values_clipped - returns) ** 2
    
    # 取较大值
    value_loss = torch.max(loss1, loss2).mean()
    
    return value_loss
```

### 精确匹配评分

```python
def normalize_answer(s: str) -> str:
    """
    标准化答案用于比较
    - 转小写
    - 去除冠词
    - 去除标点
    """
    def remove_articles(text):
        return re.sub(r'\b(a|an|the)\b', ' ', text)
    
    text = remove_articles(s.lower())
    text = re.sub(r'[^\w\s]', '', text)
    text = ' '.join(text.split())  # 规范化空格
    return text


def em_check(prediction: str, ground_truth: str) -> bool:
    """
    检查预测是否与 ground truth 精确匹配
    """
    return normalize_answer(prediction) == normalize_answer(ground_truth)


def compute_score_em(
    solution_str: str,
    ground_truth: dict,
    format_score: float = 0.,
    score: float = 1.,
) -> float:
    """
    计算精确匹配分数
    
    从 solution_str 中提取 <answer> 标签，
    与 ground_truth['target'] 比较。
    """
    # 提取答案
    answer = extract_answer(solution_str)
    
    if answer is None:
        # 没有有效答案
        return format_score
        
    # 检查是否匹配
    targets = ground_truth.get('target', [])
    for target in targets:
        if em_check(answer, target):
            return score
            
    return format_score
```
